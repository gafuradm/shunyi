def load_textbook_terms(pdf_path, textbook_id=None):
    if textbook_id is None:
        textbook_id = os.path.basename(pdf_path)

    logger.info(f"📘 加载教材: {pdf_path}")

    doc = fitz.open(pdf_path)
    text = ""

    for page in doc:
        text += page.get_text()

    candidates = re.findall(
        r'[A-Za-z]{1,10}\([^)]+\)|[A-Za-z]{2,10}|[\u4e00-\u9fff]{2,6}',
        text
    )

    freq = Counter(candidates)

    terms = {
        term for term, count in freq.items()
        if count >= 5 and len(term) <= 10
    }

    TEXTBOOKS[textbook_id] = {
        "name": os.path.basename(pdf_path),
        "terms": terms,
        "loaded_at": time.time()
    }

    ACTIVE_TEXTBOOKS.add(textbook_id)

    logger.info(f"📗 {textbook_id}: 术语数 {len(terms)}")

def float32_to_pcm16(audio):
    """将音频从 float32 转换为 PCM16"""
    if len(audio) == 0:
        return b''
    return (audio * 32767).astype(np.int16).tobytes()

def generate_session_code():
    global CURRENT_SESSION_CODE, CODE_EXPIRES_AT

    timestamp = int(time.time() / 30)
    secret = os.getenv("SESSION_SECRET", "lecture_secret_2024")

    CURRENT_SESSION_CODE = hashlib.sha256(
        f"{secret}_{timestamp}".encode()
    ).hexdigest()[:6].upper()

    CODE_EXPIRES_AT = time.time() + 30

    ACTIVE_SESSION_CODES.clear()
    ACTIVE_SESSION_CODES[CURRENT_SESSION_CODE] = {
        "expires": CODE_EXPIRES_AT,
        "used_by": []
    }

    return CURRENT_SESSION_CODE

def cleanup_old_codes():
    """删除过期代码"""
    current_time = time.time()
    expired = [code for code, data in ACTIVE_SESSION_CODES.items() 
               if data["expires"] < current_time]
    for code in expired:
        del ACTIVE_SESSION_CODES[code]

def verify_student_code(student_id, code):
    """验证学生代码"""
    code = code.upper().strip()
    
    if code not in ACTIVE_SESSION_CODES:
        return False, "代码已过期或无效"
    
    code_data = ACTIVE_SESSION_CODES[code]
    
    if time.time() > code_data["expires"]:
        del ACTIVE_SESSION_CODES[code]
        return False, "代码已过期"
    
    if student_id in code_data["used_by"]:
        return False, "该代码已被使用"
    
    code_data["used_by"].append(student_id)
    
    VERIFIED_STUDENTS[student_id] = {
        "verified": True,
        "verified_at": time.time()
    }
    
    return True, "验证成功"

def build_auth_url():
    """根据官方文档生成 iFLYTEK WebSocket 认证 URL"""
    
    # 获取 RFC1123 格式的当前日期 (GMT)
    now = datetime.now()
    date = format_date_time(mktime(now.timetuple()))
    
    # 主机 URL
    host = "ist-api-sg.xf-yun.com"
    path = "/v2/ist"
    
    # 根据文档创建签名字符串
    # 格式: host: {host}\ndate: {date}\nGET {path} HTTP/1.1
    signature_origin = f"host: {host}\ndate: {date}\nGET {path} HTTP/1.1"
    
    # 创建 HMAC-SHA256 签名
    signature_hmac = hmac.new(
        APISecret.encode('utf-8'),
        signature_origin.encode('utf-8'),
        hashlib.sha256
    ).digest()
    
    # 将签名编码为 base64
    signature_sha = base64.b64encode(signature_hmac).decode('utf-8')
    
    # 构造 authorization_origin 字符串
    authorization_origin = f'api_key="{APIKey}", algorithm="hmac-sha256", headers="host date request-line", signature="{signature_sha}"'
    
    # 将 authorization_origin 编码为 base64
    authorization = base64.b64encode(authorization_origin.encode('utf-8')).decode('utf-8')
    
    # 构建最终 URL
    url_params = {
        "host": host,
        "date": date,
        "authorization": authorization
    }
    
    # 编码 URL 参数
    query_string = urllib.parse.urlencode(url_params)
    url = f"wss://{host}{path}?{query_string}"  # Изменено с ws:// на wss://
    
    # 调试: 输出检查信息
    logger.info(f"生成 WebSocket URL (官方格式):")
    logger.info(f"  APPID: {APPID}")
    logger.info(f"  APIKey: {APIKey[:10]}...")
    logger.info(f"  APISecret: {APISecret[:10]}...")
    logger.info(f"  Date: {date}")
    logger.info(f"  Host: {host}")
    logger.info(f"  Signature origin: {signature_origin}")
    logger.info(f"  Signature SHA: {signature_sha[:50]}...")
    logger.info(f"  Authorization origin: {authorization_origin}")
    logger.info(f"  Authorization (base64): {authorization[:50]}...")
    logger.info(f"  Full URL: {url[:100]}...")
    
    return url

def on_message(ws, message):
    """处理来自 WebSocket 的消息"""
    try:
        data = json.loads(message)
        
        if data.get('code') == 0:  # 成功响应
            result_data = data.get('data', {})
            result = result_data.get('result', {})
            
            if result and 'ws' in result:
                text_parts = []
                for ws_item in result.get('ws', []):
                    for cw in ws_item.get('cw', []):
                        text_parts.append(cw.get('w', ''))
                
                text = ''.join(text_parts)
                is_final = (result_data.get('status') == 2)
                
                if text and text.strip():
                    logger.info(f"🎤 识别到 ({SOURCE_LANGUAGE}): {text}")
                    
                    # 发送原始文本给教师界面
                    socketio.emit("original_text", {
                        "text": text,
                        "is_final": is_final,
                        "source_lang": SOURCE_LANGUAGE,
                        "timestamp": time.time()
                    })
                    
                    # 翻译 для каждого клиента
                    for sid, target_lang in list(clients_lang.items()):
                        cache_key = f"{SOURCE_LANGUAGE}:{target_lang}:{text}"
                        
                        if cache_key in translation_cache:
                            translated = translation_cache[cache_key]
                        else:
                            translated = deepseek_translate(text, target_lang)
                            if translated:
                                translation_cache[cache_key] = translated
                                translation_cache.move_to_end(cache_key)
                                
                                if len(translation_cache) > MAX_CACHE:
                                    translation_cache.popitem(last=False)
                        
                        if translated:
                            socketio.emit("new_translation", {
                                "original": text,
                                "translation": translated,
                                "is_final": is_final,
                                "source_lang": SOURCE_LANGUAGE,
                                "target_lang": target_lang
                            }, to=sid)
                    
                    if is_final:
                        FULL_LECTURE_TEXT.append(text)
                        global PHRASE_COUNT
                        PHRASE_COUNT += 1
                        emit_stats()
                        
    except Exception as e:
        logger.error(f"消息处理错误: {e}")

# ================= 音频捕获线程 =================
def audio_callback(indata, frames, time_info, status):
    """音频捕获回调"""
    if status:
        logger.warning(f"音频状态: {status}")
    
    if is_running:
        try:
            # 转换为单声道并加入队列
            audio_mono = indata.copy().flatten()
            if len(audio_mono) > 0:
                audio_queue.put(audio_mono, timeout=0.1)
        except q.Full:
            pass
        except Exception as e:
            logger.error(f"audio_callback 错误: {e}")

def audio_thread():
    """麦克风音频捕获线程"""
    logger.info("🎤 启动音频捕获...")
    
    try:
        # 获取可用设备列表
        devices = sd.query_devices()
        default_device = sd.default.device[0]
        logger.info(f"使用设备: {devices[default_device]['name']}")
        
        with sd.InputStream(
            samplerate=16000,
            channels=1,
            dtype="float32",
            blocksize=8192,  # 增加以提高稳定性
            callback=audio_callback,
            device=default_device
        ):
            logger.info("✅ 麦克风就绪")
            while is_running:
                time.sleep(0.1)
                
    except Exception as e:
        logger.error(f"❌ 音频捕获错误: {e}")
        logger.error("请检查麦克风可用性和权限")

# ================= WebSocket 线程 =================
def ws_thread():
    global ws_connection, is_running

    logger.info("🌐 连接 WebSocket...")
    last_audio_time = time.time()

    while is_running:
        try:
            ws = websocket.WebSocketApp(
                build_auth_url(),
                on_open=on_open,
                on_message=on_message,
                on_error=on_error,
                on_close=on_close
            )

            wst = threading.Thread(
                target=ws.run_forever,
                kwargs={"ping_interval": 20, "ping_timeout": 10}
            )
            wst.daemon = True
            wst.start()

            logger.info("等待 WebSocket 连接...")
            time.sleep(2)

            # 🔁 主循环
            while is_running:
                try:
                    if not ws.sock or not ws.sock.connected:
                        logger.warning("WebSocket 未连接")
                        break

                    chunk = audio_queue.get(timeout=0.2)
                    last_audio_time = time.time()

                    if len(chunk) > 0:
                        chunk_pcm16 = float32_to_pcm16(chunk)

                        ws.send(json.dumps({
                            "data": {
                                "status": 1,
                                "format": "audio/L16;rate=16000",
                                "encoding": "raw",
                                "audio": base64.b64encode(chunk_pcm16).decode("utf-8")
                            }
                        }))

                except q.Empty:
                    # ⚠️ 仅在此处 continue — 在 while 内部
                    if time.time() - last_audio_time > 1.2:
                        ws.send(json.dumps({
                            "data": {
                                "status": 2,
                                "format": "audio/L16;rate=16000",
                                "encoding": "raw",
                                "audio": ""
                            }
                        }))
                    continue

                except Exception as e:
                    logger.error(f"发送音频错误: {e}")
                    break

            if is_running:
                logger.info("🔁 重启 WebSocket")
                time.sleep(1)

        except Exception as e:
            logger.error(f"❌ WebSocket 错误: {e}")
            if is_running:
                time.sleep(2)

def protect_math(text):
    return re.sub(r'([A-Za-z0-9+\-*/=<>]+)', r'`\1`', text)

def textbook_repair(text):
    for tid in ACTIVE_TEXTBOOKS:
        terms = TEXTBOOKS.get(tid, {}).get("terms", set())

        for term in terms:
            if term in text:
                continue

            if len(term) >= 3:
                pattern = re.sub('.', '.', term)
                if re.search(pattern, text):
                    text = re.sub(pattern, term, text)

    return text

Целевой язык: {target_lang_name}

Строгие правила:
- Ответ должен быть ТОЛЬКО на {target_lang_name}
- Не оставляй текст на исходном языке
- Не мешай языки

- Если текст уже на {target_lang_name}, просто исправь ошибки

Текст для перевода:
{safe_text}
"""
    
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {DEEPSEEK_API_KEY}"
    }
    
    body = {
        "model": "deepseek-chat",
        "messages": [
            {"role": "system", "content": "Ты профессиональный переводчик. Переводи точно и естественно."},
            {"role": "user", "content": prompt}
        ],
        "max_tokens": 1024,
        "temperature": 0.3
    }
    
    try:
        logger.info(f"🌐 Перевод {source_lang_name} → {target_lang_name}...")
        start_time = time.time()
        resp = requests.post(DEEPSEEK_URL, json=body, headers=headers, timeout=30)
        resp.raise_for_status()
        result = resp.json()["choices"][0]["message"]["content"].strip()
        elapsed = time.time() - start_time
        
        logger.info(f"✅ Переведено ({len(result)} символов) за {elapsed:.2f} сек")
        return result
        
    except requests.exceptions.Timeout:
        logger.error(f"❌ Таймаут перевода {source_lang_name} → {target_lang_name}")
        return text
    except requests.exceptions.RequestException as e:
        logger.error(f"❌ Ошибка запроса перевода: {e}")
        return text
    except Exception as e:
        logger.error(f"❌ Неожиданная ошибка перевода: {e}")
        return text

def emit_stats():
    socketio.emit("stats", {
        "connected": len(CONNECTED_CLIENTS),
        "phrases": PHRASE_COUNT,

@socketio.on("heartbeat")
def handle_heartbeat(data):
    sid = request.sid
    now = time.time()

    if sid not in ATTENDANCE:
        # 新学生
        ATTENDANCE[sid] = {
            "join_time": now,
            "active": 0,
            "inactive": 0,
            "last_seen": now,
        }

    a = ATTENDANCE[sid]

    if data.get("visibility") == "visible" and data.get("focused"):
        a["active"] += 5  # 5秒活跃时间
    else:
        a["inactive"] += 5

    a["last_seen"] = now

@socketio.on("connect")
def handle_connect():
    sid = request.sid
    CONNECTED_CLIENTS.add(sid)
    emit_stats()
    logger.info(f"🔗 客户端连接: {sid}")

    socketio.emit(
    "verification_required",
    {"message": "需要确认出席"},
    to=sid)

@socketio.on("disconnect")
def handle_disconnect():
    sid = request.sid
    CONNECTED_CLIENTS.discard(sid)
    emit_stats()
    logger.info(f"🔌 客户端断开: {sid}")
    clients_lang.pop(sid, None)
    ATTENDANCE.pop(sid, None)

@socketio.on("get_attendance")
def send_attendance():
    now = time.time()
    result = []
    for sid, a in ATTENDANCE.items():
        online = (now - a["last_seen"]) < 10  # 如果最近有心跳则在线上
        result.append({
            "student_id": a.get("student_id", "??"),
            "name": a.get("name", "未知"),
            "active_minutes": a.get("active", 0) // 60,
            "status": "✅" if online else "❌"
        })
    socketio.emit("attendance_update", result)

@socketio.on("set_source_language")
def handle_set_source_language(data):
    """设置源语言（老师说的语言）"""
    global SOURCE_LANGUAGE, ws_connection
    
    new_lang = data.get("lang", "zh_cn")
    
    if new_lang not in SUPPORTED_SOURCE_LANGUAGES:
        socketio.emit("source_language_error", {
            "success": False,
            "message": f"不支持的语言: {new_lang}",
            "supported_languages": list(SUPPORTED_SOURCE_LANGUAGES.keys())
        }, to=request.sid)
        return
    
    # 更新源语言
    old_lang = SOURCE_LANGUAGE
    SOURCE_LANGUAGE = new_lang
    lang_name = SUPPORTED_SOURCE_LANGUAGES[new_lang]["name"]
    
    logger.info(f"🌍 源语言切换: {old_lang} → {new_lang} ({lang_name})")
    
    # 通知所有客户端
    socketio.emit("source_language_changed", {
        "success": True,
        "lang": new_lang,
        "name": lang_name,
        "message": f"源语言已切换为 {lang_name}",
        "old_lang": old_lang
    })
    
    # 清除翻译缓存
    translation_cache.clear()
    logger.info("🧹 清除翻译缓存")
    
    # 重新启动 WebSocket 连接
    if ws_connection:
        try:
            logger.info("🔄 重启 WebSocket 连接以应用新语言...")
            ws_connection.close()
            # 等待关闭完成
            time.sleep(1)
        except Exception as e:
            logger.error(f"关闭 WebSocket 错误: {e}")

        btn.innerText = "🔊 声音开启";
        speakLast();
    } else {
        btn.innerText = "🔇 声音关闭";
        speechSynthesis.cancel();
    }
}

function speakLast() {
    if (!history.length) return;
    speak(history[0].translation);
}

let soundQueue = [];
let isSpeaking = false;

function speak(text) {
    if (!soundEnabled || !text) return;
    soundQueue.push(text);
    if (!isSpeaking) processSoundQueue();
}

function processSoundQueue() {
    if (!soundQueue.length) {
        isSpeaking = false;
        return;
    }

    isSpeaking = true;
    const text = soundQueue.shift();
    const utterance = new SpeechSynthesisUtterance(text);

    const langMap = {
        en: "en-US",
        ru: "ru-RU",
        zh: "zh-CN",
        de: "de-DE",
        fr: "fr-FR",
        es: "es-ES",
        it: "it-IT",
        ja: "ja-JP",
        ko: "ko-KR",
        pt: "pt-PT",
        ar: "ar-SA",
        hi: "hi-IN",
        tr: "tr-TR",
        pl: "pl-PL",
        uk: "uk-UA",
        kk: "ru-RU", // 哈萨克语用俄语语音
        vi: "vi-VN",
        th: "th-TH",
        ms: "ms-MY",
        id: "id-ID"
    };

    utterance.lang = langMap[selectedLang] || "en-US";
    utterance.rate = 1.0;
    utterance.pitch = 1.0;

    utterance.onend = () => {
        processSoundQueue(); // 继续下一句
    };

    speechSynthesis.speak(utterance);
}

    name = data.get("name")

        return jsonify({"error": "需要id和姓名"}), 400

    if student_id in STUDENTS:
        return jsonify({"error": "学生已注册"}), 400

    token = secrets.token_urlsafe(16)
    STUDENTS[student_id] = {"name": name, "token": token}

@app.route("/api/verified_students")
def api_verified_students():
    """获取已验证学生列表的API"""
    verified_list = []
    
    for student_id, student_data in VERIFIED_STUDENTS.items():
        if student_data["verified"]:
            student_name = "学生"
            for a in ATTENDANCE.values():
                if a.get("student_id") == student_id:
                    student_name = a.get("name", "学生")

            verified_time = datetime.fromtimestamp(
                student_data["verified_at"]
            ).strftime("%H:%M:%S")
            
            verified_list.append({
                "id": student_id,
                "name": student_name,
                "time": verified_time
            })
    
    return jsonify({
        "students": verified_list,
        "count": len(verified_list)
    })

# ================= QR码的SocketIO处理函数 =================

@socketio.on("request_verification")
def handle_verification_request():
    sid = request.sid
    logger.info(f"📩 验证请求来自 {sid}")

    socketio.emit(
        "verification_required",
        {
            "message": "需要确认出席",
            "timeout": 30
        },
        to=sid
    )

@socketio.on("verify_code")
def handle_code_verification(data):
    if not CURRENT_SESSION_CODE or time.time() > CODE_EXPIRES_AT:
        generate_session_code()
    sid = request.sid
    code = data.get("code", "").strip().upper()

    student_id = data.get("student_id")
    name = data.get("name", "学生")

    if not student_id:
        socketio.emit("verification_result", {
            "success": False,
            "message": "缺少student_id"
        }, to=sid)
        return

    success, message = verify_student_code(student_id, code)

    if success:
        if sid not in ATTENDANCE:
            ATTENDANCE[sid] = {}

        ATTENDANCE[sid]["student_id"] = student_id
        ATTENDANCE[sid]["name"] = name

    socketio.emit("verification_result", {
        "success": success,
        "message": message
    }, to=sid)
