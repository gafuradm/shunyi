# ================= Вспомогательные функции =================
def float32_to_pcm16(audio):
    """Конвертация аудио из float32 в PCM16"""
    if len(audio) == 0:
        return b''
    return (audio * 32767).astype(np.int16).tobytes()

def build_auth_url():
    """Генерация URL для аутентификации WebSocket iFLYTEK согласно официальной документации"""
    
    # Получаем текущую дату в формате RFC1123 (GMT)
    now = datetime.now()
    date = format_date_time(mktime(now.timetuple()))
    
    # URL хоста
    host = "ist-api-sg.xf-yun.com"
    path = "/v2/ist"
    
    # Создаем строку для подписи согласно документации
    # Формат: host: {host}\ndate: {date}\nGET {path} HTTP/1.1
    signature_origin = f"host: {host}\ndate: {date}\nGET {path} HTTP/1.1"
    
    # Создаем HMAC-SHA256 подпись
    signature_hmac = hmac.new(
        APISecret.encode('utf-8'),
        signature_origin.encode('utf-8'),
        hashlib.sha256
    ).digest()
    
    # Кодируем подпись в base64
    signature_sha = base64.b64encode(signature_hmac).decode('utf-8')
    
    # Формируем строку authorization_origin
    authorization_origin = f'api_key="{APIKey}", algorithm="hmac-sha256", headers="host date request-line", signature="{signature_sha}"'
    
    # Кодируем authorization_origin в base64
    authorization = base64.b64encode(authorization_origin.encode('utf-8')).decode('utf-8')
    
    # Формируем финальный URL
    url_params = {
        "host": host,
        "date": date,
        "authorization": authorization
    }
    
    # Кодируем параметры для URL
    query_string = urllib.parse.urlencode(url_params)
    url = f"ws://{host}{path}?{query_string}"
    
    # ДЕБАГ: Выведем информацию для проверки
    logger.info(f"Генерация URL для WebSocket (официальный формат):")
    logger.info(f"  APPID: {APPID}")
    logger.info(f"  APIKey: {APIKey[:10]}...")
    logger.info(f"  APISecret: {APISecret[:10]}...")
    logger.info(f"  Date: {date}")
    logger.info(f"  Host: {host}")
    logger.info(f"  Signature origin: {signature_origin}")
    logger.info(f"  Signature SHA: {signature_sha[:50]}...")
    logger.info(f"  Authorization origin: {authorization_origin}")
    logger.info(f"  Authorization (base64): {authorization[:50]}...")
    logger.info(f"  Full URL: {url[:100]}...")
    
    return url

# ================= WebSocket обработчики =================
def on_message(ws, message):
    """Обработка сообщений от WebSocket"""
    try:
        data = json.loads(message)
        logger.info(f"Получено сообщение с кодом: {data.get('code')}")
        
        if data.get('code') == 0:  # Успешный ответ
            result_data = data.get('data', {})
            result = result_data.get('result', {})
            
            # Проверяем наличие текста в результате
            if result and 'ws' in result:
                # Извлекаем текст из структуры ws
                text_parts = []
                for ws_item in result['ws']:
                    for cw in ws_item.get('cw', []):
                        text_parts.append(cw.get('w', ''))
                
                text = ''.join(text_parts)
                
                is_final = (result_data.get('status') == 2)

                if text:
                    for sid, lang in list(clients_lang.items()):
                        cache_key = f"{lang}:{text}"
                        if cache_key in translation_cache:
                            translated = translation_cache[cache_key]
                        else:
                            translated = deepseek_translate(text, lang)
                            translation_cache[cache_key] = translated

                        socketio.emit("new_translation", {
                            "original": text,
                            "translation": translated,
                            "is_final": is_final
                        }, to=sid)

                    if is_final:
                        FULL_LECTURE_TEXT.append(text)
                            
        elif data.get('code') != 0:
            logger.error(f"Ошибка от сервера: {data.get('message')}")
            
    except Exception as e:
        logger.error(f"Ошибка обработки сообщения: {e}")

def on_error(ws, error):
    """Обработка ошибок WebSocket"""
    logger.error(f"WebSocket ошибка: {error}")

def on_close(ws, close_status_code, close_msg):
    """Обработка закрытия WebSocket"""
    logger.info(f"WebSocket закрыт: {close_msg}")

def on_open(ws):
    """Обработка открытия WebSocket"""
    global ws_connection
    logger.info("✅ WebSocket подключен")
    ws_connection = ws
    
    # Отправляем начальные параметры согласно документации
    init_params = {
        "common": {
            "app_id": APPID  # Здесь используется APPID
        },
        "business": {
            "language": "zh_cn",  # Китайский язык
            "domain": "ist_ed",  # Домен для распознавания
            "accent": "mandarin",  # Мандаринский акцент
            "punc": 1,  # Включаем пунктуацию
            "nunum": 1   # Включаем форматирование чисел
        },
        "data": {
            "status": 0,  # Первый фрейм
            "format": "audio/L16;rate=16000",
            "encoding": "raw",
            "audio": ""  # Пустой для первого фрейма
        }
    }
    
    ws.send(json.dumps(init_params))
    logger.info("Отправлены init параметры")

# ================= Поток аудиозахвата =================
def audio_callback(indata, frames, time_info, status):
    """Callback для захвата аудио"""
    if status:
        logger.warning(f"Аудио статус: {status}")
    
    if is_running:
        try:
            # Конвертируем в моно и добавляем в очередь
            audio_mono = indata.copy().flatten()
            if len(audio_mono) > 0:
                audio_queue.put(audio_mono, timeout=0.1)
        except q.Full:
            pass
        except Exception as e:
            logger.error(f"Ошибка в audio_callback: {e}")

def audio_thread():
    """Поток для захвата аудио с микрофона"""
    logger.info("🎤 Запуск аудиозахвата...")
    
    try:
        # Получаем список доступных устройств
        devices = sd.query_devices()
        default_device = sd.default.device[0]
        logger.info(f"Используется устройство: {devices[default_device]['name']}")
        
        with sd.InputStream(
            samplerate=16000,
            channels=1,
            dtype="float32",
            blocksize=8192,  # Увеличим для стабильности
            callback=audio_callback,
            device=default_device
        ):
            logger.info("✅ Микрофон готов")
            while is_running:
                time.sleep(0.1)
                
    except Exception as e:
        logger.error(f"❌ Ошибка аудиозахвата: {e}")
        logger.error("Проверьте доступность микрофона и разрешения")

# ================= Поток WebSocket =================
def ws_thread():
    global ws_connection, is_running

    logger.info("🌐 Подключение к WebSocket...")
    last_audio_time = time.time()

    while is_running:
        try:
            ws = websocket.WebSocketApp(
                build_auth_url(),
                on_open=on_open,
                on_message=on_message,
                on_error=on_error,
                on_close=on_close
            )

            wst = threading.Thread(
                target=ws.run_forever,
                kwargs={"ping_interval": 20, "ping_timeout": 10}
            )
            wst.daemon = True
            wst.start()

            logger.info("Ожидание подключения WebSocket...")
            time.sleep(2)

            # 🔁 ГЛАВНЫЙ ЦИКЛ
            while is_running:
                try:
                    if not ws.sock or not ws.sock.connected:
                        logger.warning("WebSocket не подключен")
                        break

                    chunk = audio_queue.get(timeout=0.2)
                    last_audio_time = time.time()

                    if len(chunk) > 0:
                        chunk_pcm16 = float32_to_pcm16(chunk)

                        ws.send(json.dumps({
                            "data": {
                                "status": 1,
                                "format": "audio/L16;rate=16000",
                                "encoding": "raw",
                                "audio": base64.b64encode(chunk_pcm16).decode("utf-8")
                            }
                        }))

                except q.Empty:
                    # ⚠️ continue ТОЛЬКО ЗДЕСЬ — внутри while
                    if time.time() - last_audio_time > 1.2:
                        ws.send(json.dumps({
                            "data": {
                                "status": 2,
                                "format": "audio/L16;rate=16000",
                                "encoding": "raw",
                                "audio": ""
                            }
                        }))
                    continue

                except Exception as e:
                    logger.error(f"Ошибка отправки аудио: {e}")
                    break

            if is_running:
                logger.info("🔁 Перезапуск WebSocket")
                time.sleep(1)

        except Exception as e:
            logger.error(f"❌ Ошибка WebSocket: {e}")
            if is_running:
                time.sleep(2)

def protect_math(text):
    return re.sub(r'([A-Za-z0-9+\-*/=<>]+)', r'`\1`', text)

# ================= Semantic repair =================
def semantic_repair(text):
    prompt = f"""
Ты ассистент преподавателя университета.

Твоя задача:
Определи, есть ли в тексте ошибки распознавания речи (ASR),
которые искажают математический смысл.

ЕСЛИ ошибок НЕТ:
верни текст БЕЗ ИЗМЕНЕНИЙ.

ЕСЛИ ошибки ЕСТЬ:
верни исправленный текст, используя стандартную математическую терминологию.

Правила:
- НЕ добавляй новых идей
- НЕ объясняй
- НЕ улучшай стиль
- НЕ переводи
- Возвращай ТОЛЬКО текст

Текст:
{text}
"""
    resp = requests.post(
        DEEPSEEK_URL,
        headers={
            "Authorization": f"Bearer {DEEPSEEK_API_KEY}",
            "Content-Type": "application/json",
        },
        json={
            "model": "deepseek-chat",
            "messages": [
                {
                    "role": "system",
                    "content": "Ты автоматически исправляешь ASR ошибки в математических лекциях."
                },
                {
                    "role": "user",
                    "content": prompt
                }
            ],
            "temperature": 0.0
        },
        timeout=15,
    )

    return resp.json()["choices"][0]["message"]["content"].strip()

# ================= Переводчик DeepSeek =================
def deepseek_translate(text, target_lang):
    """Перевод текста через DeepSeek API"""

    # 1️⃣ Семантическое исправление
    try:
        repaired = semantic_repair(text)
        if repaired:
            text = repaired
    except Exception as e:
        logger.warning(f"Semantic repair skipped: {e}")

    if not text.strip():
        return ""
    
    if not DEEPSEEK_API_KEY or DEEPSEEK_API_KEY == "your_deepseek_api_key_here":
        logger.warning("⚠️ DEEPSEEK_API_KEY не установлен или установлен по умолчанию")
        return f"[Перевод: {text}]"
    
    # 2️⃣ Маппинг языка
    lang_names = {
        "en": "английский", "de": "немецкий", "kk": "казахский",
        "ru": "русский", "ja": "японский", "ko": "корейский",
        "vi": "вьетнамский", "th": "тайский", "ms": "малайский",
        "id": "индонезийский", "mn": "монгольский", "hy": "армянский",
        "it": "итальянский", "fr": "французский", "es": "испанский"
    }
    
    target_lang_name = lang_names.get(target_lang, target_lang)
    
    safe_text = protect_math(text)
    prompt = f"""Ты переводишь университетскую лекцию по математике.
Делай перевод естественным академическим английским
Переведи следующий текст на {target_lang_name}.
Передавай только перевод, без пояснений.
Текст: {safe_text}"""
    
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {DEEPSEEK_API_KEY}"
    }
    
    body = {
        "model": "deepseek-chat",
        "messages": [
            {"role": "system", "content": "Ты профессиональный переводчик."},
            {"role": "user", "content": prompt}
        ],
        "max_tokens": 1024,
        "temperature": 0.3
    }
    
    try:
        logger.info(f"🌐 Перевод на {target_lang}...")
        resp = requests.post(DEEPSEEK_URL, json=body, headers=headers, timeout=30)
        resp.raise_for_status()
        result = resp.json()["choices"][0]["message"]["content"].strip()
        logger.info(f"✅ Переведено ({len(result)} символов)")
        return result
    except Exception as e:
        logger.error(f"❌ Ошибка перевода: {e}")
        return text

# ================= SocketIO обработчики =================
@socketio.on("connect")
def handle_connect():
    """Обработка подключения клиента"""
    sid = request.sid
    logger.info(f"🔗 Клиент подключен: {sid}")
    # Устанавливаем язык по умолчанию
    clients_lang[sid] = "en"

@socketio.on("disconnect")
def handle_disconnect():
    """Обработка отключения клиента"""
    sid = request.sid
    if sid in clients_lang:
        del clients_lang[sid]
    logger.info(f"🔌 Клиент отключен: {sid}")

@socketio.on("set_language")
def handle_set_language(data):
    """Установка языка перевода"""
    lang = data.get("lang", "en")
    sid = request.sid
    clients_lang[sid] = lang
    logger.info(f"🌍 Клиент {sid[:8]}... выбрал язык: {lang}")
    socketio.emit("language_set", {"lang": lang}, to=sid)

@socketio.on("get_full_text")
def send_full_text():
    """Отправка полного текста лекции"""
    sid = request.sid
    text = "\n".join(FULL_LECTURE_TEXT[-50:])  # Последние 50 строк
    socketio.emit("full_text", {"text": text}, to=sid)

# ================= Graceful shutdown =================
def signal_handler(sig, frame):
    """Обработка сигналов завершения"""
    global is_running
    logger.info("\n🛑 Получен сигнал завершения...")
    is_running = False
    time.sleep(1)
    logger.info("✅ Сервер остановлен")
    sys.exit(0)

# ================= Запуск =================
if __name__ == "__main__":
    # Проверка API ключей
    if not APIKey or not APISecret:
        logger.warning("⚠️ Внимание: API ключи iFLYTEK не настроены!")
        logger.warning("   Используются тестовые ключи. Для продакшена получите свои на xfyun.cn")
    
    if not DEEPSEEK_API_KEY or DEEPSEEK_API_KEY == "your_deepseek_api_key_here":
        logger.warning("⚠️ Внимание: DEEPSEEK_API_KEY не настроен!")
        logger.warning("   Получите ключ на https://platform.deepseek.com/ и добавьте в .env файл")
    
    # Регистрация обработчиков сигналов
    signal.signal(signal.SIGINT, signal_handler)
    signal.signal(signal.SIGTERM, signal_handler)
    
    # Запуск потоков
    logger.info("🚀 Запуск сервера...")
    
    # Запускаем аудио поток
    audio_thread_obj = threading.Thread(target=audio_thread, daemon=True)
    audio_thread_obj.start()
    
    # Запускаем WebSocket поток
    ws_thread_obj = threading.Thread(target=ws_thread, daemon=True)
    ws_thread_obj.start()
    
    # Даем время на инициализацию
    time.sleep(2)
    
    logger.info("✅ Сервер запущен!")
    logger.info("👨‍🏫 Преподаватель: http://localhost:8000")
    logger.info("🎓 Студент: http://localhost:8000/student")
    logger.info("")
    logger.info("🎤 Говорите в микрофон для начала перевода...")
    logger.info("🛑 Для остановки нажмите Ctrl+C")
    
    # Запуск Flask приложения
    try:
        socketio.run(app, 
                    host="0.0.0.0", 
                    port=8000,
                    debug=False,
                    allow_unsafe_werkzeug=True)
    except KeyboardInterrupt:
        signal_handler(None, None)
