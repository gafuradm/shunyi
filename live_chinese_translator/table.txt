        function blockAllKeys(e) {
            const key = e.key.toLowerCase();
            // Блокируем всё, что связано со скриншотами, печатью, копированием, F12
            if (e.ctrlKey || e.metaKey || e.altKey || 
                key === 'printscreen' || 
                e.keyCode === 44 ||        // PrintScreen
                e.keyCode === 123 ||        // F12
                key.includes('f12') ||
                key === 'f12' ||
                (e.ctrlKey && (key === 'p' || key === 's' || key === 'c' || key === 'v'))) {
                e.preventDefault();
                e.stopPropagation();
                
                // Показываем предупреждение
