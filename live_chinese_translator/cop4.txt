def load_textbook_terms(pdf_path, textbook_id=None):
    if textbook_id is None:
        textbook_id = os.path.basename(pdf_path)

    logger.info(f"📘 加载教材: {pdf_path}")

    doc = fitz.open(pdf_path)
    text = ""

    for page in doc:
        text += page.get_text()

    # Улучшенное извлечение математических терминов
    candidates = re.findall(
        r'[A-Za-z]{1,10}\([^)]+\)|'  # f(x) стиль
        r'[A-Za-z]{2,10}|'            # переменные
        r'[\u4e00-\u9fff]{2,6}|'      # китайские символы
        r'[0-9]+[a-zA-Z]+|'            # числа с переменными
        r'[a-zA-Z]+\^[0-9]+|'          # степени
        r'√[a-zA-Z0-9]+|'              # корни
        r'∫|∑|∏|lim|sin|cos|tan|log|ln', # мат. символы
        text
    )

    freq = Counter(candidates)

    terms = {
        term for term, count in freq.items()
        if count >= 3 and len(term) <= 15  # Увеличил длину для формул
    }
    
    # Добавляем стандартные математические термины
    terms.update(MATH_TERMS_CN.values())

    TEXTBOOKS[textbook_id] = {
        "name": os.path.basename(pdf_path),
        "terms": terms,
        "loaded_at": time.time()
    }

    ACTIVE_TEXTBOOKS.add(textbook_id)

    logger.info(f"📗 {textbook_id}: 术语数 {len(terms)} (包括数学公式)")

def float32_to_pcm16(audio):
    """将音频从 float32 转换为 PCM16"""
    if len(audio) == 0:
        return b''
    return (audio * 32767).astype(np.int16).tobytes()

def generate_session_code():
    global CURRENT_SESSION_CODE, CODE_EXPIRES_AT

    timestamp = int(time.time() / 45)
    secret = os.getenv("SESSION_SECRET", "lecture_secret_2024")

    CURRENT_SESSION_CODE = hashlib.sha256(
        f"{secret}_{timestamp}".encode()
    ).hexdigest()[:6].upper()

    CODE_EXPIRES_AT = time.time() + 45

    ACTIVE_SESSION_CODES.clear()
    ACTIVE_SESSION_CODES[CURRENT_SESSION_CODE] = {
        "expires": CODE_EXPIRES_AT,
        "used_by": []
    }

    return CURRENT_SESSION_CODE

def cleanup_old_codes():
    """删除过期代码"""
    current_time = time.time()
    expired = [code for code, data in ACTIVE_SESSION_CODES.items() 
               if data["expires"] < current_time]
    for code in expired:
        del ACTIVE_SESSION_CODES[code]

def verify_student_code(student_id, code):
    """验证学生代码"""
    code = code.upper().strip()
    
    if code not in ACTIVE_SESSION_CODES:
        return False, "代码已过期或无效"
    
    code_data = ACTIVE_SESSION_CODES[code]
    
    if time.time() > code_data["expires"]:
        del ACTIVE_SESSION_CODES[code]
        return False, "代码已过期"
    
    if student_id in code_data["used_by"]:
        return False, "该代码已被使用"
    
    code_data["used_by"].append(student_id)
    
    VERIFIED_STUDENTS[student_id] = {
        "verified": True,
        "verified_at": time.time()
    }
    
    return True, "验证成功"

def build_auth_url():
    """根据官方文档生成 iFLYTEK WebSocket 认证 URL"""
    
    # 获取 RFC1123 格式的当前日期 (GMT)
    now = datetime.now()
    date = format_date_time(mktime(now.timetuple()))
    
    # 主机 URL
    host = "ist-api-sg.xf-yun.com"
    path = "/v2/ist"
    
    # 根据文档创建签名字符串
    # 格式: host: {host}\ndate: {date}\nGET {path} HTTP/1.1
    signature_origin = f"host: {host}\ndate: {date}\nGET {path} HTTP/1.1"
    
    # 创建 HMAC-SHA256 签名
    signature_hmac = hmac.new(
        APISecret.encode('utf-8'),
        signature_origin.encode('utf-8'),
        hashlib.sha256
    ).digest()
    
    # 将签名编码为 base64
    signature_sha = base64.b64encode(signature_hmac).decode('utf-8')
    
    # 构造 authorization_origin 字符串
    authorization_origin = f'api_key="{APIKey}", algorithm="hmac-sha256", headers="host date request-line", signature="{signature_sha}"'
    
    # 将 authorization_origin 编码为 base64
    authorization = base64.b64encode(authorization_origin.encode('utf-8')).decode('utf-8')
    
    # 构建最终 URL
    url_params = {
        "host": host,
        "date": date,
        "authorization": authorization
    }
    
    # 编码 URL 参数
    query_string = urllib.parse.urlencode(url_params)
    url = f"ws://{host}{path}?{query_string}"
    
    # 调试: 输出检查信息
    logger.info(f"生成 WebSocket URL (官方格式):")
    logger.info(f"  APPID: {APPID}")
    logger.info(f"  APIKey: {APIKey[:10]}...")
    logger.info(f"  APISecret: {APISecret[:10]}...")
    logger.info(f"  Date: {date}")
    logger.info(f"  Host: {host}")
    logger.info(f"  Signature origin: {signature_origin}")
    logger.info(f"  Signature SHA: {signature_sha[:50]}...")
    logger.info(f"  Authorization origin: {authorization_origin}")
    logger.info(f"  Authorization (base64): {authorization[:50]}...")
    logger.info(f"  Full URL: {url[:100]}...")
    
    return url

def on_message(ws, message):
    """处理来自 WebSocket 的消息"""
    try:
        data = json.loads(message)
        logger.info(f"收到代码消息: {data.get('code')}")
        
        if data.get('code') == 0:  # 成功响应
            result_data = data.get('data', {})
            result = result_data.get('result', {})
            
            # 检查结果中是否有文本
            if result and 'ws' in result:
                # 从 ws 结构中提取文本
                text_parts = []
                for ws_item in result['ws']:
                    for cw in ws_item.get('cw', []):
                        text_parts.append(cw.get('w', ''))
                
                text = ''.join(text_parts)
                
                is_final = (result_data.get('status') == 2)

                if text:
                    for sid, lang in list(clients_lang.items()):
                        cache_key = f"{lang}:{text}"
                        if cache_key in translation_cache:
                            translated = translation_cache[cache_key]
                        else:
                            translated = deepseek_translate(text, lang)
                            translation_cache[cache_key] = translated
                            translation_cache.move_to_end(cache_key)

                            if len(translation_cache) > MAX_CACHE:
                                translation_cache.popitem(last=False)

                        socketio.emit("new_translation", {
                            "original": text,
                            "translation": translated,
                            "is_final": is_final
                        }, to=sid)

                    if is_final:
                        FULL_LECTURE_TEXT.append(text)

                        global PHRASE_COUNT
                        PHRASE_COUNT += 1
                        emit_stats()
                            
        elif data.get('code') != 0:
            logger.error(f"服务器错误: {data.get('message')}")
            
    except Exception as e:
        logger.error(f"消息处理错误: {e}")

def audio_callback(indata, frames, time_info, status):
    """音频捕获回调"""
    if status:
        logger.warning(f"音频状态: {status}")
    
    if is_running:
        try:
            # 转换为单声道并加入队列
            audio_mono = indata.copy().flatten()
            if len(audio_mono) > 0:
                audio_queue.put(audio_mono, timeout=0.1)
        except q.Full:
            pass
        except Exception as e:
            logger.error(f"audio_callback 错误: {e}")

def audio_thread():
    """麦克风音频捕获线程"""
    logger.info("🎤 启动音频捕获...")
    
    try:
        # 获取可用设备列表
        devices = sd.query_devices()
        default_device = sd.default.device[0]
        logger.info(f"使用设备: {devices[default_device]['name']}")
        
        with sd.InputStream(
            samplerate=16000,
            channels=1,
            dtype="float32",
            blocksize=8192,  # 增加以提高稳定性
            callback=audio_callback,
            device=default_device
        ):
            logger.info("✅ 麦克风就绪")
            while is_running:
                time.sleep(0.1)
                
    except Exception as e:
        logger.error(f"❌ 音频捕获错误: {e}")
        logger.error("请检查麦克风可用性和权限")

# ================= WebSocket 线程 =================
def ws_thread():
    global ws_connection, is_running

    logger.info("🌐 连接 WebSocket...")
    last_audio_time = time.time()

    while is_running:
        try:
            ws = websocket.WebSocketApp(
                build_auth_url(),
                on_open=on_open,
                on_message=on_message,
                on_error=on_error,
                on_close=on_close
            )

            wst = threading.Thread(
                target=ws.run_forever,
                kwargs={"ping_interval": 20, "ping_timeout": 10}
            )
            wst.daemon = True
            wst.start()

            logger.info("等待 WebSocket 连接...")
            time.sleep(2)

            # 🔁 主循环
            while is_running:
                try:
                    if not ws.sock or not ws.sock.connected:
                        logger.warning("WebSocket 未连接")
                        break

                    chunk = audio_queue.get(timeout=0.2)
                    last_audio_time = time.time()

                    if len(chunk) > 0:
                        chunk_pcm16 = float32_to_pcm16(chunk)

                        ws.send(json.dumps({
                            "data": {
                                "status": 1,
                                "format": "audio/L16;rate=16000",
                                "encoding": "raw",
                                "audio": base64.b64encode(chunk_pcm16).decode("utf-8")
                            }
                        }))

                except q.Empty:
                    # ⚠️ 仅在此处 continue — 在 while 内部
                    if time.time() - last_audio_time > 1.2:
                        ws.send(json.dumps({
                            "data": {
                                "status": 2,
                                "format": "audio/L16;rate=16000",
                                "encoding": "raw",
                                "audio": ""
                            }
                        }))
                    continue

                except Exception as e:
                    logger.error(f"发送音频错误: {e}")
                    break

            if is_running:
                logger.info("🔁 重启 WebSocket")
                time.sleep(1)

        except Exception as e:
            logger.error(f"❌ WebSocket 错误: {e}")
            if is_running:
                time.sleep(2)

def process_math_formulas(text):
    """处理中文数学公式转换为标准符号"""
    # 首先替换完整短语（更长的优先）
    phrases = sorted(MATH_TERMS_CN.keys(), key=len, reverse=True)
    
    for phrase in phrases:
        if phrase in text:
            text = text.replace(phrase, MATH_TERMS_CN[phrase])
    
    # 处理常见的数学表达式模式
    patterns = [
        # 处理 x 的 n 次方 -> x^n
        (r'([a-zA-Z0-9πe])\s*的\s*([0-9]+)\s*次方', r'\1^\2'),
        (r'([a-zA-Z0-9πe])\s*的\s*平方', r'\1²'),
        (r'([a-zA-Z0-9πe])\s*的\s*立方', r'\1³'),
        
        # 处理根号
        (r'根号\s*([a-zA-Z0-9πe]+)', r'√(\1)'),
        (r'([0-9]+)\s*次根号\s*([a-zA-Z0-9πe]+)', r'\1√\2'),
        
        # 处理分数
        (r'([a-zA-Z0-9πe]+)\s*分之\s*([a-zA-Z0-9πe]+)', r'\2/\1'),
        
        # 处理下标
        (r'([a-zA-Z])\s*下标\s*([0-9a-zA-Z])', r'\1_\2'),
        
        # 处理极限
        (r'当\s*([a-zA-Z0-9]+)\s*趋于\s*([0-9∞]+)', r'lim_{\1→\2}'),
        
        # 处理积分
        (r'从\s*([0-9a-zA-Z]+)\s*到\s*([0-9a-zA-Z]+)\s*的积分', r'∫_{\1}^{\2}'),
    ]
    
    for pattern, replacement in patterns:
        text = re.sub(pattern, replacement, text)
    
    return text

def protect_math(text):
    return re.sub(r'([A-Za-z0-9+\-*/=<>]+)', r'`\1`', text)

def textbook_repair(text):
    # Сначала обрабатываем математические формулы
    text = process_math_formulas(text)
    
    for tid in ACTIVE_TEXTBOOKS:
        terms = TEXTBOOKS.get(tid, {}).get("terms", set())
        
        # Добавляем математические термины к терминам из учебника
        math_terms = set(MATH_TERMS_CN.values())
        all_terms = terms.union(math_terms)
        
        for term in all_terms:
            if term in text:
                continue
            if len(term) >= 2:
                # Умный поиск похожих терминов
                if term in text or term.lower() in text.lower():
                    continue
                    
    return text

def deepseek_translate(text, target_lang):
    """通过 DeepSeek API 翻译文本"""
    # 📘 根据教材修复
    try:
        text = textbook_repair(text)
    except Exception as e:
        logger.warning(f"教材修复跳过: {e}")

    # 1️⃣ 语义修复
    try:
        repaired = semantic_repair(text)
        if repaired:
            text = repaired
    except Exception as e:
        logger.warning(f"语义修复跳过: {e}")

    if not text.strip():
        return ""
    
    if not DEEPSEEK_API_KEY or DEEPSEEK_API_KEY == "your_deepseek_api_key_here":
        logger.warning("⚠️ DEEPSEEK_API_KEY 未设置或为默认值")
        return f"[翻译: {text}]"
    
    # 2️⃣ 语言映射
    lang_names = {
        "en": "英语", "de": "德语", "kk": "哈萨克语",
        "ru": "俄语", "ja": "日语", "ko": "韩语",
        "vi": "越南语", "th": "泰语", "ms": "马来语",
        "id": "印尼语", "mn": "蒙古语", "hy": "亚美尼亚语",
        "it": "意大利语", "fr": "法语", "es": "西班牙语"
    }
    
    target_lang_name = lang_names.get(target_lang, target_lang)
    
    safe_text = protect_math(text)
    prompt = f"""
你是专业翻译。

严格规则：
- 回答必须仅使用语言: {target_lang_name}
- 不留其他语言的文本
- 不保留原始语言
- 不混合语言
- 如果文本已经是 {target_lang_name}，仍然重述它

仅返回翻译。

文本：
{safe_text}
"""
    
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {DEEPSEEK_API_KEY}"
    }
    
    body = {
        "model": "deepseek-chat",
        "messages": [
            {"role": "system", "content": "你是专业翻译。"},
            {"role": "user", "content": prompt}
        ],
        "max_tokens": 1024,
        "temperature": 0.3
    }
    
    try:
        logger.info(f"🌐 翻译到 {target_lang}...")
        resp = requests.post(DEEPSEEK_URL, json=body, headers=headers, timeout=30)
        resp.raise_for_status()
        result = resp.json()["choices"][0]["message"]["content"].strip()
        logger.info(f"✅ 已翻译 ({len(result)} 字符)")
        return result
    except Exception as e:
        logger.error(f"❌ 翻译错误: {e}")
        return text

@socketio.on("heartbeat")
def handle_heartbeat(data):
    sid = request.sid
    now = time.time()

    if sid not in ATTENDANCE:
        # 新学生
        ATTENDANCE[sid] = {
            "join_time": now,
            "active": 0,
            "inactive": 0,
            "last_seen": now,
            "verified": False  # Добавляем флаг верификации
        }

    a = ATTENDANCE[sid]
    
    # Проверяем, верифицирован ли студент
    student_id = a.get("student_id")
    if student_id and student_id in VERIFIED_STUDENTS:
        a["verified"] = True

    # Всегда считаем студента активным, если он верифицирован
    # Игнорируем видимость экрана
    if a.get("verified", False):
        a["active"] += 5  # Всегда добавляем активность
    else:
        # Для неверифицированных используем старую логику
        if data.get("visibility") == "visible" and data.get("focused"):
            a["active"] += 5
        else:
            a["inactive"] += 5

    a["last_seen"] = now

@socketio.on("connect")

@socketio.on("disconnect")
def handle_disconnect():

    CONNECTED_CLIENTS.discard(sid)
    emit_stats()
    logger.info(f"🔌 客户端断开: {sid}")
    clients_lang.pop(sid, None)
    ATTENDANCE.pop(sid, None)

@socketio.on("get_attendance")
def send_attendance():
    now = time.time()
    result = []
    for sid, a in ATTENDANCE.items():
        # Увеличиваем таймаут для верифицированных студентов
        timeout = 30 if a.get("verified", False) else 10
        online = (now - a["last_seen"]) < timeout
        
        # Определяем статус
        if a.get("verified", False):
            status = "✅ (фон)" if online else "❌"
        else:
            status = "✅" if online else "❌"
            
        result.append({
            "student_id": a.get("student_id", "??"),
            "name": a.get("name", "未知"),
            "active_minutes": a.get("active", 0) // 60,
            "status": status,
            "verified": a.get("verified", False)
        })
    socketio.emit("attendance_update", result)

@socketio.on("student_active")

    student_id = data.get("student_id")
    
    if sid in ATTENDANCE and student_id:
        ATTENDANCE[sid]["last_seen"] = time.time()
        ATTENDANCE[sid]["active"] += 5

@socketio.on("restore_session")
def handle_restore_session(data):
    sid = request.sid
    student_id = data.get("student_id")
    
    # Проверяем, был ли студент верифицирован ранее
    if student_id and student_id in VERIFIED_STUDENTS:
        # Восстанавливаем сессию
        if sid not in ATTENDANCE:
            ATTENDANCE[sid] = {}
        
        ATTENDANCE[sid]["student_id"] = student_id
        ATTENDANCE[sid]["verified"] = True
        ATTENDANCE[sid]["restored_at"] = time.time()
        ATTENDANCE[sid]["last_seen"] = time.time()
        
        socketio.emit("session_restored", {
            "success": True,
            "message": "Сессия восстановлена"
        }, to=sid)
    else:
        socketio.emit("session_restored", {
            "success": False,
            "message": "Сессия не найдена"
        }, to=sid)

@app.route("/api/verified_students")
def api_verified_students():
    """获取已验证学生列表的API"""
    verified_list = []
    
    for student_id, student_data in VERIFIED_STUDENTS.items():
        if student_data["verified"]:
            student_name = "学生"
            for a in ATTENDANCE.values():
                if a.get("student_id") == student_id:
                    student_name = a.get("name", "学生")

            verified_time = datetime.fromtimestamp(
                student_data["verified_at"]
            ).strftime("%H:%M:%S")
            
            verified_list.append({
                "id": student_id,
                "name": student_name,
                "time": verified_time
            })
    
    return jsonify({
        "students": verified_list,
        "count": len(verified_list)
    })

    if success:

        ATTENDANCE[sid]["name"] = f"学生_{student_id[-4:]}"  # Генерируем имя из ID

    socketio.emit("verification_result", {
        "success": success,

    signal.signal(signal.SIGINT, signal_handler)
    signal.signal(signal.SIGTERM, signal_handler)
    
    # 启动线程
    logger.info("🚀 启动服务器...")
    
    # 启动音频线程
    audio_thread_obj = threading.Thread(target=audio_thread, daemon=True)
    audio_thread_obj.start()
    
    # 启动WebSocket线程
    ws_thread_obj = threading.Thread(target=ws_thread, daemon=True)
    ws_thread_obj.start()

    code_thread = threading.Thread(target=code_refresh_task, daemon=True)
    code_thread.start()
    
    # 给予初始化时间
    time.sleep(2)
    
    logger.info("✅ 服务器已启动!")
    logger.info("👨‍🏫 教师: http://localhost:8000")
    logger.info("🎓 学生: http://localhost:8000/student")
    logger.info("")
    logger.info("🎤 请对着麦克风讲话开始翻译...")
    logger.info("🛑 按Ctrl+C停止")
    
    # 启动Flask应用
    try:
        socketio.run(app, 
                    host="0.0.0.0", 
                    port=8000,
                    debug=False,
                    allow_unsafe_werkzeug=True)
    except KeyboardInterrupt:
        signal_handler(None, None)
